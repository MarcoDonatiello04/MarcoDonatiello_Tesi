% !TEX root = ../main.tex
% Diagramma UML di attivita' della funzione analyze() del modulo di analisi
% del codice applicativo per la Security Misconfiguration.
\linespread{1.0}\selectfont
\begin{tikzpicture}[
  font=\footnotesize,
  >=Stealth,
  node distance=0.55cm and 0.8cm,
  action/.style={draw, rounded corners=6pt, align=center, inner xsep=7pt,
                 inner ysep=5pt, fill=white, text width=4.2cm},
  decision/.style={draw, diamond, aspect=2.2, inner sep=1.5pt, align=center,
                   font=\scriptsize, fill=white},
  initial/.style={circle, fill=black, minimum size=8pt, inner sep=0pt},
  final/.style={circle, draw, minimum size=11pt, inner sep=0pt,
                path picture={\fill[black] (path picture bounding box.center)
                              circle (3.2pt);}},
  bar/.style={draw, fill=black, minimum width=9.0cm, minimum height=2.5pt,
              inner sep=0pt},
  guard/.style={font=\scriptsize, fill=white, inner sep=1.5pt},
  flow/.style={->}
]
% --- Colonna sinistra: ciclo sui file ---------------------------------------
\node[initial] (start) {};
\node[action, below=of start] (walk)
  {Attraversa il repository\\ {\scriptsize salta \texttt{.git}, \texttt{.venv}, \texttt{node\_modules},
   \texttt{\_\_pycache\_\_}, \texttt{dist}, \texttt{build}}};
\node[decision, below=of walk] (more) {altro file\\ \texttt{.py}/\texttt{.js}/\texttt{.ts}?};
\node[action, below=0.7cm of more] (read) {Leggi il sorgente};
\node[decision, below=of read] (ispy) {file \texttt{.py}?};
\node[action, below=0.7cm of ispy] (parse) {Costruisci l'AST\\ {\scriptsize modulo \texttt{ast}}};
\node[decision, below=of parse] (parsed) {AST valido?};
\node[bar, below=0.7cm of parsed] (fork) {};
% regole locali in parallelo
\node[action, below=0.5cm of fork, xshift=-3.45cm, text width=1.6cm] (r1) {SC-001\\ \scriptsize CORS};
\node[action, right=0.2cm of r1, text width=1.6cm] (r2) {SC-002\\ \scriptsize debug};
\node[action, right=0.2cm of r2, text width=1.6cm] (r3) {SC-003\\ \scriptsize errori};
\node[action, right=0.2cm of r3, text width=1.6cm] (r5) {SC-005\\ \scriptsize segreti};
\node[bar, anchor=north] (join) at ($(fork.south |- r2.south)+(0,-0.5)$) {};
\node[action, below=of join, text width=5.4cm] (collect) {Accoda le segnalazioni del file};
% --- Binari laterali per i ritorni al ciclo ---------------------------------
\coordinate (railL) at ($(r1.west)+(-0.4,0)$);
\coordinate (railR) at ($(r5.east)+(0.4,0)$);
% --- Colonna destra: chiusura -----------------------------------------------
\node[action, anchor=west, text width=4.4cm] (global) at ($(railR |- more)+(0.6,0)$)
  {Regola globale SC-004\\ {\scriptsize al più una segnalazione per repository}};
\node[action, below=of global, text width=4.4cm] (filter)
  {Scarta le segnalazioni\\ con confidenza $< 0{,}70$};
\node[action, below=of filter, text width=4.4cm] (report)
  {Componi il \texttt{MisconfigReport}\\ {\scriptsize segnalazioni, segnali di copertura, riepilogo}};
\node[final, below=of report] (end) {};
% --- Flussi -----------------------------------------------------------------
\draw[flow] (start) -- (walk);
\draw[flow] (walk) -- (more);
\draw[flow] (more) -- node[guard, right] {[sì]} (read);
\draw[flow] (read) -- (ispy);
\draw[flow] (ispy) -- node[guard, right] {[sì]} (parse);
\draw[flow] (parse) -- (parsed);
\draw[flow] (parsed) -- node[guard, right] {[sì]} (fork);
\draw[flow] (fork.south -| r1) -- (r1);
\draw[flow] (fork.south -| r2) -- (r2);
\draw[flow] (fork.south -| r3) -- (r3);
\draw[flow] (fork.south -| r5) -- (r5);
\draw[flow] (r1) -- (join.north -| r1);
\draw[flow] (r2) -- (join.north -| r2);
\draw[flow] (r3) -- (join.north -| r3);
\draw[flow] (r5) -- (join.north -| r5);
\draw[flow] (join) -- (collect);
% ritorno al ciclo dopo il file
\draw[flow] (collect.west) -- (railL |- collect) -- (railL |- more) -- (more.west);
% AST non valido: il file viene saltato
\draw (parsed.west) -- node[guard, above] {[no]} (railL |- parsed);
% file .js/.ts: nessun AST, le regole vengono comunque invocate (solo SC-001 opera, su regex)
\draw[flow] (ispy.east) -- node[guard, above] {[no]} (railR |- ispy)
  -- node[guard, left, align=right] {\texttt{.js}/\texttt{.ts}:\\ solo SC-001\\ (regex)} (railR |- fork) -- (fork.east);
% fine del ciclo sui file
\draw[flow] (more.east) -- node[guard, above] {[no]} (global.west);
\draw[flow] (global) -- (filter);
\draw[flow] (filter) -- (report);
\draw[flow] (report) -- (end);
\end{tikzpicture}
