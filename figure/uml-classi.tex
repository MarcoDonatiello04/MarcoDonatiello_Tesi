\begin{class}[text width=4.6cm]{Finding}{6.3,0}
  \attribute{finding\_id: str}
  \attribute{source: FindingSource}
  \attribute{category: FindingCategory}
  \attribute{severity: Severity}
  \attribute{confidence: float}

\begin{class}[text width=2.8cm]{CodeLocation}{1.2,1.7}
  \attribute{file\_path: str}

\end{class}
\begin{class}[text width=2.8cm]{APIContext}{1.2,-1.6}
  \attribute{endpoint: str}
  \attribute{method: str}

\end{class}
\begin{class}[text width=3.9cm]{RuntimeEvidence}{12.9,1.7}
  \attribute{tested\_url: str}
  \attribute{http\_status: int}
  \attribute{response\_headers: dict}
\end{class}
\begin{class}[text width=3.9cm]{RiskContext}{12.9,-1.6}

  \attribute{sensitive\_data\_detected: bool}

\end{class}
\composition{Finding}{location}{0..1}{CodeLocation}

% --- Enumerazioni ----------------------------------------------------------
\begin{class}[text width=2.6cm]{Severity}{1.2,-6.0}
  \attribute{CRITICAL = 9.0}
  \attribute{HIGH = 7.0}
  \attribute{MEDIUM = 4.5}
  \attribute{LOW = 2.0}
  \attribute{INFO}
  \operation{score(): float}
\end{class}
\begin{class}[text width=2.8cm]{ValidationStatus}{4.3,-6.0}
  \attribute{NOT\_VALIDATED}
  \attribute{CONFIRMED}
\end{class}
\unidirectionalAssociation{Finding}{}{1}{Severity}
\unidirectionalAssociation{Finding}{}{1}{ValidationStatus}
% --- Motore di correlazione ------------------------------------------------

  \operation{calculate\_risk\_score(f): float}
