% Diagramma UML delle classi: entita' Finding con i suoi oggetti valore,
% le enumerazioni Severity e FindingNature, l'interfaccia IScanner con
% l'adapter Checkov e il suo catalogo delle policy, e il motore di correlazione. Il disegno e' su
% un canvas largo e viene ridotto alla larghezza del testo.
\linespread{1.0}\selectfont
\resizebox{\linewidth}{!}{%
\begin{tikzpicture}[font=\scriptsize]
\def\umldrawcolor{black}
\def\umlfillcolor{white}
\def\umltextcolor{black}
% --- Entita' di dominio e oggetti valore -----------------------------------
\begin{class}[text width=4.6cm]{Finding}{6.3,0}
  \attribute{finding\_id: str}
  \attribute{source: FindingSource}
  \attribute{category: FindingCategory}
  \attribute{severity: Severity}
  \attribute{confidence: float}
  \attribute{rule\_id, cwe\_id: str}
  \attribute{resource\_id: str}
  \attribute{correlation\_key: str}
  \attribute{related\_findings: list[str]}
  \attribute{nature: FindingNature}
  \operation{create(\ldots): Finding}
  \operation{nature\_rank(): int}
  \operation{is\_directly\_exploitable(): bool}
\end{class}
\begin{class}[text width=2.8cm]{CodeLocation}{1.2,1.7}
  \attribute{file\_path: str}
  \attribute{start\_line: int}
  \attribute{end\_line: int}
\end{class}
\begin{class}[text width=2.8cm]{APIContext}{1.2,-1.6}
  \attribute{endpoint: str}
  \attribute{method: str}
  \attribute{base\_url: str}
\end{class}
\begin{class}[text width=3.9cm]{RiskContext}{12.9,0.2}
  \attribute{internet\_exposed: bool}
  \attribute{sensitive\_data\_detected: bool}
  \attribute{public\_resource: bool}
\end{class}
\composition{Finding}{location}{0..1}{CodeLocation}
\composition{Finding}{api}{0..1}{APIContext}
\composition{Finding}{risk\_context}{0..1}{RiskContext}
% --- Enumerazione ----------------------------------------------------------
\begin{class}[text width=2.6cm]{Severity}{1.2,-6.0}
  \attribute{CRITICAL = 9.0}
  \attribute{HIGH = 7.0}
  \attribute{MEDIUM = 4.5}
  \attribute{LOW = 2.0}
  \attribute{INFO}
  \operation{score(): float}
\end{class}
\begin{class}[text width=2.9cm]{FindingNature}{4.6,-6.0}
  \attribute{EXPOSURE}
  \attribute{HARDENING}
  \operation{rank(): int}
\end{class}
\unidirectionalAssociation{Finding}{}{1}{Severity}
\unidirectionalAssociation{Finding}{}{0..1}{FindingNature}
% --- Motore di correlazione ------------------------------------------------
\begin{class}[text width=5.0cm]{RiskCorrelationEngine}{9.1,-6.0}
  \attribute{correlated\_findings: dict[str, Finding]}
  \operation{correlate(static, runtime): list[Finding]}
  \operation{calculate\_risk\_score(f): float}
  \operation{\_outranks(candidate, incumbent): bool}
\end{class}
\begin{staticclass}[text width=3.6cm]{APIEndpointNormalizer}{9.1,-9.6}
  \operation{normalize\_path(path): str}
\end{staticclass}
\unidirectionalAssociation{RiskCorrelationEngine}{}{*}{Finding}
\draw[umlcd style, dashed, ->] (RiskCorrelationEngine) -- (APIEndpointNormalizer)
  node[midway, right, font=\scriptsize]{\guillemotleft use\guillemotright};
% --- Interfaccia scanner e adapter -----------------------------------------
\begin{interface}[text width=3.9cm]{IScanner}{14.0,-6.0}
  \operation{scan(target\_dir): list[Finding]}
\end{interface}
\draw[umlcd style, dashed, ->] (IScanner) -- (Finding)
  node[pos=0.3, above, sloped, font=\scriptsize]{\guillemotleft create\guillemotright};
\begin{class}[text width=3.9cm]{CheckovScannerAdapter}{14.0,-9.4}
  \operation{scan(target\_dir): list[Finding]}
  \operation{build\_finding(check): Finding}
\end{class}
\draw[umlcd style, dashed, ->, >=open triangle 45] (CheckovScannerAdapter) -- (IScanner);
\begin{class}[text width=4.7cm]{CheckovPolicyCatalog}{14.2,-12.6}
  \attribute{policies, prefixes, keywords}
  \operation{classify(id, name): PolicyClassification}
\end{class}
\begin{class}[text width=3.4cm]{PolicyClassification}{8.6,-12.6}
  \attribute{nature: FindingNature}
  \attribute{severity: Severity}
  \attribute{public\_facing: bool}
  \attribute{matched\_by: str}
\end{class}
\unidirectionalAssociation{CheckovScannerAdapter}{policy\_catalog}{1}{CheckovPolicyCatalog}
\draw[umlcd style, dashed, ->] (CheckovPolicyCatalog) -- (PolicyClassification)
  node[midway, above, font=\scriptsize]{\guillemotleft create\guillemotright};
\end{tikzpicture}%
}
